# argentina-income-analyzer.pages.dev: Vite copies web/public/ into web/dist/.
# Search: yes. AI training: no. Same policy as the daddiego.com.ar sites.
#
# Until this file existed, /robots.txt fell through to the Pages SPA fallback and
# answered 200 with index.html AND Content-Type: text/html, so crawlers were
# parsing the app shell as robots rules.
#
# This project has no custom domain, so it is served straight off *.pages.dev,
# where Cloudflare's zone-level "managed robots.txt" does not apply. Unlike the
# sites on daddiego.com.ar, this file is served exactly as written.

User-agent: *
Content-Signal: search=yes,ai-train=no,use=reference
Allow: /

# Nothing is disallowed on purpose: the page renders client-side from the data
# artifact (/assets/percentiles.v1-<hash>.json), so a crawler that cannot fetch
# that file sees an empty shell.
# Blocking the data artifact would cost the search listing, not protect it.

# No AI model training. The AI *search* bots (OAI-SearchBot, Claude-SearchBot,
# PerplexityBot) are left out on purpose: being cited there is fine.
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: CCBot
User-agent: Google-Extended
User-agent: Bytespider
User-agent: Amazonbot
User-agent: Applebot-Extended
User-agent: meta-externalagent
Disallow: /

# No Sitemap: line. The site is a single page, so the root URL is the sitemap.
